const FIREBASE_URL = "https://vmteam-fcb3e-default-rtdb.firebaseio.com"; const SHORTENER_URL = "https://prospexurl.com.br/r/getkey1"; async function clicarPegarKey() { const btn = document.getElementById("getKeyBtn"); if (btn.disabled) return; btn.disabled = true; btn.classList.add("loading"); btn.textContent = "Validando acesso..."; try { const response = await fetch("https://api.ipify.org?format=json"); if (!response.ok) throw new Error("Não foi possível obter o IP."); const data = await response.json(); if (!data.ip) throw new Error("IP inválido."); const ip = data.ip; const safeIP = ip.replace(/[.#$[\]/]/g, "_"); const save = await fetch( `${FIREBASE_URL}/visitas/ips/${safeIP}.json`, { method: "PUT", headers: {"Content-Type":"application/json"}, body: JSON.stringify({ ip: ip, status: "pendente", createdAt: Date.now() }) } ); if (!save.ok) throw new Error("Erro ao registrar acesso."); localStorage.setItem("vtx_ip", ip); localStorage.setItem("vtx_country", "BR"); btn.textContent = "Acesso autorizado!"; window.location.href = SHORTENER_URL; } catch (error) { console.error(error); btn.disabled = false; btn.classList.remove("loading"); btn.textContent = "Tentar novamente"; alert(error.message); } }
